FIR filter
static float const firCoefficients[]=
{
	//adjust these values accordingly
	0.f,0.f,0.f,0.1f,0.2f,0.3f,0.4f
};

float Initialize(void)
{
	for(int i = 0;i < mHistoryBuffer.size();++i) mCoefficients[i] = firCoefficents[i]l
}

float Update(float const input)//input can be camera distance to target or other values
{
	//copy the entries in the history buffer up by one positin
	for(int i = mHistoryBuffer.size() -2;i >=0;--i)mHistoryBuffer[i + 1] = mHistoryBuffer[i];

	mHistoryBuffer[0] = input;//record new value as-is
	float fir(0);
	for(int i = 0;i < mHistoryBuffer.size();++i) fir += mCoefficients[i] * mHistoryBuffer[i];
	return fir;
}

IIR filter (incorporates output values)
void Initialize(void)
{
	mInputCoefficients[0] = 0.5f;
	mInputCoefficients[1] = 0.3f;
	mOutputCoefficients[0] = 0.5f;
	mOutputCoefficients[1] = 0.3f;
}

float Update(float const input)
{
	for(int i = mInputHistoryBuffer.size() - 2;i >= 0;--i) mInputHistoryBuffer[i+1]=mInputHistoryBuffer[i];
	mInputHistoryBuffer[0] = input;
float const result = mInputCoefficients[0] * mInputHistoryBuffer[0] + mInputCoefficients[1] * mInputHistoryBuffer[1] + mOutputCoefficients[0]*mOutputHistoryBuffer[0] + mOutputCoefficients[1]*mOutputHistoryBuffer[1]
for(int i=mOutputHistoryBuffer.size() - 2;i >=0;--i) mOutputHistoryBuffer[i + 1]=mOutputHistoryBuffer[i];
mOutputHistoryBuffer[0] = result;
return result;
}
